cmake_minimum_required(VERSION 3.24)
project(tracy-crop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TRACY_SOURCE_DIR)
  message(FATAL_ERROR "Set TRACY_SOURCE_DIR to the Tracy 0.14.1 source directory")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py"
  "${TRACY_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated"
  COMMAND_ERROR_IS_FATAL ANY)
set(NO_STATISTICS ON)
find_package(zstd CONFIG REQUIRED)
# Offline file processing does not need live disassembly or parallel sorting.
add_library(crop-capstone INTERFACE)
add_library(capstone::capstone ALIAS crop-capstone)
add_library(crop-sort INTERFACE)
target_include_directories(crop-sort INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/compat")
add_library(PPQSort::PPQSort ALIAS crop-sort)
include("${TRACY_SOURCE_DIR}/cmake/server.cmake")
list(REMOVE_ITEM TRACY_SERVER_SOURCES "${TRACY_SERVER_DIR}/TracyWorker.cpp")
set_property(TARGET TracyServer PROPERTY SOURCES
  "${TRACY_COMMON_SOURCES};${TRACY_SERVER_SOURCES};${CMAKE_CURRENT_BINARY_DIR}/generated/TracyWorker.cpp")
target_compile_definitions(TracyServer PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
target_compile_options(TracyServer PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(tracy-crop main.cpp)
target_include_directories(tracy-crop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(tracy-crop PRIVATE TracyServer)
target_compile_options(tracy-crop PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
